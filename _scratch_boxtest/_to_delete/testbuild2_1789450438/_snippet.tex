\begin{frame}{Aktivitet}

  \aktivitet{Akselerasjon av et skip}{%
  Et skip beveger seg rettlinjet og er paavirket av en motorkraft $F_m = XXX\,\mathrm{kN}$ framover og en motstandskraft fra boelgene $F_b = YYY\,\mathrm{kN}$ bakover. Skipets masse er $m = ZZZ\,\mathrm{tonn}$.
  \begin{enumerate}[a)]
      \item Finn resultantkraften paa skipet.
      \item Beregn skipets akselerasjon $a$.
  \end{enumerate}
  }

\end{frame}

\begin{frame}{Aktivitet}

  \aktivitet{Terminalfart}{%
  En stein har naadd terminalfart paa vei ned mot bakken. Paa steinen virker tyngdekraften $G = 6\,\mathrm{N}$ (nedover) og luftmotstanden $R = 6\,\mathrm{N}$ (oppover).

  Akselererer steinen? Begrunn svaret.
  }

  \pause
  \aktivitet{Lampa i taket}{%
  Paa ei lampe som henger i ro virker tyngdekraften (nedover) og snorkraften (oppover). Tyngdekraften er $G = 10\,\mathrm{N}$.
  \begin{enumerate}[a)]
      \item Tegn inn kreftene som virker paa lampa.
      \item Hva er snorkrafta?
      \item Er snorkrafta motkrafta til $G$?
  \end{enumerate}
  }

\end{frame}
