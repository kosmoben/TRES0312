%--------------------------------------------------------------------------
%  SLIDE 4 — Definisjon: Hookes lov
%--------------------------------------------------------------------------
\begin{frame}{Kraft fra spiralfjær}

  \defnat{4.6}{Hookes lov}{
  Kraften fra en fjær på det den er festet til er proporsjonal med avviket $\vec{x}$ fra likevektsposisjonen:
  \[
    \vec{F} = -k\vec{x}
  \]
  Her er $k>0$ \textit{fjærkonstanten} (enhet: $\mathrm{N/m}$), og $\vec{x}$ er forflytningen fra likevekt. Minustegnet viser at kraften alltid peker \textit{mot} likevekt.
  }

\end{frame}

      \defnat{4.7}{Tyngdekraft}{
      \footnotesize
      Tyngdekraften $\vec{G}$ er kraften jorda trekker et legeme med masse $m$ mot sitt sentrum med:
      \[
        \vec{G} = m\vec{\mathrm{g}}, \qquad \mathrm{g} \approx 9.81\,\mathrm{m/s^2}.

      \footnotesize

\begin{frame}{Underlagskrefter}

  Når et objekt står på et underlag, dekomponerer vi underlagskraften $\vec{U}$ i normalkraft $\vec{N}$ (vinkelrett på underlaget) og friksjonskraft $\vec{R}$ (parallelt med underlaget):
  \[
    \vec{U} = \vec{N} + \vec{R}

%--------------------------------------------------------------------------
%  SLIDE 13 — Definisjon: Kinetisk friksjonskoeffisient
%--------------------------------------------------------------------------
\begin{frame}{Friksjon}

  \defnat{4.8}{Kinetisk friksjonskoeffisient}{
  Den \textbf{kinetiske} friksjonskoeffisienten $\mu_k$ bestemmer glidefriksjonen:
  \[R_k = \mu_k N.\]
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 14 — Aktivitet: Boks som sklir (glidefriksjon)
%--------------------------------------------------------------------------
\begin{frame}{Aktivitet}

  \aktivitet{Boks som sklir}{%
  En boks med masse $m = 10\,\mathrm{kg}$ glir på et gulv med kinetisk friksjonskoeffisient $\mu_k = 0.30$.
  \begin{enumerate}[a)]
      \item Finn friksjonskraften $R_k$ som virker på boksen.
      \item Friksjonskraften er den eneste horisontale kraften. Finn boksens akselerasjon.
      \item Er akselerasjonen konstant?
  \end{enumerate}
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 15 — Definisjon: Statisk friksjonskoeffisient
%--------------------------------------------------------------------------
\begin{frame}{Friksjon}

  \defnat{4.9}{Statisk friksjonskoeffisient}{
  Den \textbf{statiske} friksjonskoeffisienten $\mu_s$ bestemmer maksimal hvilefriksjon:
  \[R_s \leq \mu_s N.\]
  }

  \merk{}{Som regel er $\mu_k < \mu_s$.}

\end{frame}
