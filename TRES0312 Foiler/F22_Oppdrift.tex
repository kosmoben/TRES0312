%--------------------------------------------------------------------------
%  SLIDE 3 — Til ettertanke: steiner under vann
%--------------------------------------------------------------------------
\begin{frame}{Oppdrift}

  \tanke{}{%
  Hvorfor føles en stein lettere når du løfter den under vann?
  }
  \pause
  \svar{
  Trykket i vannet øker med dybden. Derfor presser vannet hardere opp på undersiden av steinen enn ned på oversiden. Resultatet er en kraft oppover: \textit{oppdriftskraften}.
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 3b — Hvor kommer oppdriften fra?
%--------------------------------------------------------------------------
\begin{frame}{Oppdrift}

  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      Trykket øker med dybden, så væsken presser hardere opp på bunnen enn ned på toppen:
      \begin{itemize}
        \item<2-> $F_\text{topp} = (p_0 + \rho_v\mathrm{g}h_1)\,A$
        \item<2-> $F_\text{bunn} = (p_0 + \rho_v\mathrm{g}h_2)\,A$
      \end{itemize}
      \onslide<3->{Nettokraften oppover er \textit{oppdriftskraften} $F_{\textrm{B}}$ (B for engelsk \textit{buoyancy}):
      \[
        F_{\textrm{B}} = F_\text{bunn} - F_\text{topp} = \rho_v\mathrm{g}A h_s = \rho_v\mathrm{g}V_\text{f}
      \]}
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering
      \begin{tikzpicture}[scale=1.0, font=\small, >={Stealth[length=7pt,width=5pt]}]
        \fill[oppgcol!30!pagecolor] (-1.5,-0.4) rectangle (3.0,2.5);

        \draw[<->, thin, textcolor!70] (2.7,1.9) -- (2.7,0.6) node[midway, right, font=\footnotesize] {$h_s$};

%--------------------------------------------------------------------------
%  SLIDE 4 — Definisjon: oppdriftskraft
%--------------------------------------------------------------------------
\begin{frame}{Oppdrift}

  \defnat{7.4.1}{Arkimedes' oppdriftslov}{
  Oppdriftskraften på et legeme i en væske i ro er
  \[
    F_{\textrm{B}} = \rho_v\,\mathrm{g}\,V_\text{f}
  \]
  der $\rho_v$ er tettheten til væsken og $V_\text{f}$ er volumet av væsken som legemet fortrenger. Dette tilsvarer tyngden av den fortrengte væsken.
  }

  \pause
  \merk{Fortrengt volum}{%
  $V_\text{f}$ er den delen av legemet som er under væskeoverflaten. Er legemet helt under, er $V_\text{f}$ lik hele volumet.
  }

\end{frame}

\begin{frame}{Flyter eller synker?}

  \tanke{}{%
  Stål har nesten åtte ganger så stor tetthet som vann. Hvordan kan et skip av stål flyte?
  }
  \pause
  \svar{
  Det er skipet \textit{som helhet} som teller: stål pluss all lufta inni. Den gjennomsnittlige tettheten er mindre enn tettheten til vann.
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 9 — Flyter eller synker?
%--------------------------------------------------------------------------
\begin{frame}{Flyter eller synker?}

  For et legeme helt under væsken er $G = \rho_L\,\mathrm{g}\,V$ og $F_{\textrm{B}} = \rho_v\,\mathrm{g}\,V$:
  \graabox{$\displaystyle F_\text{netto} = F_{\textrm{B}} - G = (\rho_v - \rho_L)\,\mathrm{g}\,V$}
  \pause
  \begin{itemize}
    \item $\rho_L < \rho_v$: legemet \textbf{flyter}.
    \item $\rho_L = \rho_v$: legemet \textbf{svever}.
    \item $\rho_L > \rho_v$: legemet \textbf{synker}.
  \end{itemize}
  \pause
  Når et legeme flyter i ro, er $F_{\textrm{B}} = G$.

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 12c — Til ettertanke: masse eller volum?
%--------------------------------------------------------------------------
\begin{frame}{Flyter eller synker?}

  \tanke{}{%
  Hvorfor stiger vannet mer når steinen ligger i båten?
  }
  \pause
  \svar{
  En stein som flyter (i båten), må fortrenge vann med like stor \textit{masse} som seg selv. En stein på bunnen fortrenger bare vann med like stort \textit{volum} som seg selv. Siden steinen har større tetthet enn vann, er det siste mindre vann.
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 13 — Oppsummering: flyter eller synker
%--------------------------------------------------------------------------
\begin{frame}{}

  \oppsum{Flyter eller synker?}{%
  \begin{enumerate}[1.]
    \item[] Sammenlign tettheten til legemet med tettheten til væsken.
    \item[] Et legeme som flyter i ro, har $F_{\textrm{B}} = G$.
    \item[] Oppdriften er lik tyngden av den fortrengte væsken (Arkimedes' oppdriftslov).
  \end{enumerate}
  }

\end{frame}
