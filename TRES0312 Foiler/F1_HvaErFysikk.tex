\begin{frame}{}
  
  \spq{Hva er fysikk?}
  \pause
  \svar{
  Fysikk er studiet av naturens grunnstruktur, gjerne med matematikk som verktøy.
  }
\pause
  \tanke{}{%
    Kan alt i verden beskrives i matematikk? Hva med \textit{kjærlighet} eller \textit{bevissthet}? Slike vanskelige, filosofiske spørsmål skal vi ikke vsare på i dette kompendiet, men de er verdt å merke seg.}
 
\end{frame}

  }

\begin{frame}{Oppgave}
  \oppgavexr{enhetsregning}{Enhetsregning}{%
    En bil kjører med fart $v = 90\,\textrm{km/t}$.
    \begin{enumerate}[a)]
      \item Gjør om farten til m/s. \textit{Hint: } $1\,\mathrm{km} = 10^{3}\,\mathrm{m}$ og $1\,\mathrm{t} = 3600\,\mathrm{s}$.
      \item Kjøreturen varer i $t = 2{,}5\,\mathrm{t}$. Finn tilbakelagt strekning i km og i m.
      \item Tyngdeakselerasjonen på månen er $g_{\mathrm{M}} = 1{,}62\,\mathrm{m/s}^{2}$. Gjør om til $\mathrm{km/min}^{2}$.
    \end{enumerate}
  }
\end{frame}
