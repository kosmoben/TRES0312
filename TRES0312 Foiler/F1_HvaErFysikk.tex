\begin{frame}{}
  
  \spq{Hva er fysikk?}
  \pause
  \svar{
  Fysikk er studiet av naturens grunnstruktur, gjerne med matematikk som verktøy.
  }
\pause
  \tanke{}{%
    Kan alt i verden beskrives i matematikk? Hva med \textit{kjærlighet} eller \textit{bevissthet}? Slike vanskelige, filosofiske spørsmål skal vi ikke vsare på i dette kompendiet, men de er verdt å merke seg.}
 
\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 3 — FYSISKE STØRRELSER
%--------------------------------------------------------------------------

\begin{frame}{Fysiske størrelser}
  \spq{Hvordan beskriver vi virkeligheten ved hjelp av matematikk?}
  \pause
  \svar{
    \begin{itemize}
      \item Vi bruker fysiske størrelser.
      \item Vi må være nøyaktige
    \end{itemize}
  }
  \pause
  \defnxr{fys-storrelse}{Fysisk størrelse}{%
    En \textit{fysisk størrelse} har både \textit{verdi} og \textit{enhet}:
    \[
      \textrm{Fysisk størrelse} = \textrm{verdi} \cdot \textrm{enhet}.
    \]
  }

\end{frame}
