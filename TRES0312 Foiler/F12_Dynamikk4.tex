%--------------------------------------------------------------------------
%  SLIDE 3 — Definisjon: Sentripetalkraft
%--------------------------------------------------------------------------
\begin{frame}{Sentripetalkraft}

  Fra kinematikken vet vi at sentripetalakselerasjonen i sirkelbevegelse med konstant fart er $a_\perp = v^2/r$. Newtons 2. lov, $\sum\vec{F}=m\vec{a}$, gir da et uttrykk for kraftsummen inn mot sentrum.

  \pause
  \defnat{4.10}{Sentripetalkraft}{
  Kraftsummen for sirkelbevegelse med radius $r$ og konstant banefart $v$ er gitt ved
  \[\sum F = m\cdot\frac{v^2}{r},\]
  og er rettet inn mot sentrum av sirkelbevegelsen.
  }

\end{frame}

\begin{frame}{Dosert veibane}

  \tanke{}{%
  Kan en bil kjøre gjennom en sirkelformet sving \textit{uten friksjon} mellom dekk og vei?
  }

  \pause
  Ja --- hvis veibanen er dosert (hellet innover) med riktig vinkel $\alpha$. Da gir normalkraftens horisontale komponent sentripetalakselerasjonen.

\end{frame}

\begin{frame}{Eksempel: dosering av veibane}

  \exmat{4.13}{Dosering av veibane}{%
  Er det mulig for en bil med konstant fart $v$ å kjøre gjennom en sirkelformet sving med radius $r$ uten friksjon mellom dekk og vei? Bestem nødvendig dosering (helningsvinkel) $\alpha$. Bruk $v = 90\,\mathrm{km/h}$ og $r = 200\,\mathrm{m}$.
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 12 — Løsning: dosering av veibane
%--------------------------------------------------------------------------
\begin{frame}{}

  Dekomponering ($V$ vertikalt, $H$ inn mot sentrum), ingen friksjon:
  \[
    N_V = m\mathrm{g}, \qquad N_H = m\frac{v^2}{r}
  \]

  \pause
  Fra geometrien er $\tan\alpha = N_H/N_V$:
  \[
    \tan\alpha = \frac{v^2}{r\,\mathrm{g}}
  \]

  \pause
  \graabox{$\displaystyle \alpha = \arctan\!\left(\frac{v^2}{r\,\mathrm{g}}\right) = \arctan(0.318) = \underline{\underline{17.7^\circ}}$}

  \pause
  \merk{Uavhengig av masse}{
  Massen faller ut av likningen --- doseringen er uavhengig av kjøretøyets masse.
  }

\end{frame}

\begin{frame}{}

  \oppsum{Krefter i sirkelbevegelse}{%
  \begin{enumerate}[1.]
    \item[] Sentripetalkraft: $\sum F = m v^2/r$, rettet inn mot sentrum.
    \item[] Dosert veibane (uten friksjon): $\tan\alpha = v^2/(r\,\mathrm{g})$.
  \end{enumerate}
