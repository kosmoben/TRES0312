      \begin{pycode}
import numpy as np
import matplotlib.pyplot as plt

g, h0 = 9.81, 28.25     # m/s^2, m
m, k = 0.670, 0.00145   # kg, kg/m (se notatbok)

def simuler(k, dt=1e-4):
    t = v = h = 0.0
    ts, hs = [0.0], [0.0]
    while h < h0:
        a = g - (k/m)*v*abs(v)
        v += a*dt
        h += v*dt
        t += dt
        ts.append(t)
        hs.append(h)
    return np.array(ts), np.array(hs)

t_u, h_u = simuler(k=0)   # uten luftmotstand
t_m, h_m = simuler(k=k)   # med luftmotstand

plt.plot(t_u, h_u, "--", label="Uten luftmotstand")
plt.plot(t_m, h_m, "-", label="Med luftmotstand")
plt.xlabel("Tid [s]")
plt.ylabel("Fallhøyde [m]")
plt.legend()
plt.show()
      \end{pycode}
