%  SLIDE 11 — Funksjoner: introduksjon
%--------------------------------------------------------------------------
\begin{frame}{Funksjoner og plotting}

  \defn{Funksjon}{
  En \textit{funksjon} er en regel som til enhver innverdi $x$ knytter én bestemt utverdi $f(x)$. I fysikk er posisjonen $s$ til et objekt i bevegelse for eksempel en funksjon av tid $t$, og vi skriver da $s(t)$.
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 12 — Lineær funksjon
%--------------------------------------------------------------------------
\begin{frame}{Lineær funksjon}

  \defn{Lineær funksjon}{
  \[
    f(x) = b + ax,
  \]
  der $a$ er \textit{stigningstallet} (hvor bratt grafen er) og $b$ er \textit{konstantleddet} (der grafen skjærer $y$-aksen).
  }
  \pause
  \svar{
  Et eksempel er farten $v(t)$ ved konstant akselerasjon:
  \[
    v(t) = v_0 + at.
  \]
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 13 — Kvadratisk funksjon
%--------------------------------------------------------------------------
\begin{frame}{Kvadratisk funksjon}

  \defn{Kvadratisk funksjon}{
  \[
    f(x) = ax^2 + bx + c.
  \]
  Grafen er en \textit{parabel}.
  }
  \pause
  \svar{
  Posisjonen til et legeme med konstant akselerasjon er et eksempel på en kvadratisk funksjon av tid:
  \[
    s(t) = \tfrac{1}{2}at^2 + v_0 t + s_0.
  \]
  }

\end{frame}

\begin{frame}{Trigonometri}

  Trigonometri er et svært nyttig redskap i fysikken, ettersom vi hele veien har bruk for å dekomponere vektorer i et koordinatsystem.

  \begin{center}
  \begin{tikzpicture}[scale=1.05, baseline=(current bounding box.center)]
    \coordinate (O) at (0,0);
    \coordinate (A) at (3.0,0);
    \coordinate (B) at (3.0,1.9);
    \fill[headCol!12] (O) -- (A) -- (B) -- cycle;
    \draw[thick] (O) -- (A) node[midway, below=6pt] {\small hosliggende ($a$)};
    \draw[thick] (A) -- (B) node[midway, right=5pt] {\small motstående ($b$)};
    \draw[thick] (O) -- (B) node[midway, above left=2pt] {\small hypotenus ($c$)};
    \draw (A) ++(-0.2,0) -- ++(0,0.2) -- ++(0.2,0);
    \draw[headCol, thick] (0.5,0) arc (0:32:0.5);
    \node at (0.8,0.13) {$\theta$};
  \end{tikzpicture}
  \end{center}

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 18 — Definisjon: sinus, cosinus og tangens
%--------------------------------------------------------------------------
\begin{frame}{Sinus, cosinus og tangens}

  \defn{Sinus, cosinus og tangens}{
  For en vinkel $\theta$ i en rettvinklet trekant med hypotenus $c$:
  \[
    \cos\theta = \frac{\text{hosligg.}}{\text{hyp.}} = \frac{a}{c}, \qquad
    \sin\theta = \frac{\text{motst.}}{\text{hyp.}} = \frac{b}{c}, \qquad
    \tan\theta = \frac{\sin\theta}{\cos\theta} = \frac{b}{a}.
  \]
  }

\end{frame}

\begin{frame}{Enhetssirkelen}

  Enhetssirkelen er en sirkel med radius 1. Katetene i trekanten med vinkel $\theta$ tegnet inn i enhetssirkelen blir da nettopp $\cos\theta$ og $\sin\theta$. Fra Pythagoras' læresetning ser vi da umiddelbart at:

  \graabox{%
    $\cos^2\theta + \sin^2\theta = 1$
  }

  \pause
  \tanke{}{%
  Denne formelen er mye brukt --- den kan du bare lære deg med det samme.
  }

\end{frame}
