%--------------------------------------------------------------------------
%  SLIDE 3 — Definisjon: Effekt
%--------------------------------------------------------------------------
\begin{frame}{Effekt}

  Effekt $P$ er et mål på hvor raskt energi omformes, eller hvor raskt arbeid utføres.

  \pause
  \defnat{5.7}{Effekt som arbeid per tidsenhet}{
  \[
    P = \frac{W}{t}
  \]
  Enheten for effekt er watt ($\mathrm{W}$), der $1\,\mathrm{W} = 1\,\mathrm{J/s}$.
  }

\end{frame}

%  SLIDE 4 — Definisjon: Effekt
%--------------------------------------------------------------------------
\begin{frame}{Aktivitet}

  \aktivitet{Ovn}{%
  En 1200 Watts ovn står på i en time. Hvor mye energi har den trukket?
  }
  \aktivitet{Bil}{%
  En bil med masse $m=1200\,\mathrm{kg}$ akselererer fra $0$ til $100\,\mathrm{km/t}$ på $10.0\,\mathrm{s}$.

  Hvor stor effekt må motoren yte for å oppnå denne akselerasjonen?
  }
  \aktivitet{Mann}{%
  En voksen person forbrenner rundt 2000 kcal per dag. Hvor stor effekt tilsvarer dette? Hvor mange 1000-watts varmeovner utgjør dere som er i forelesning totalt sett nå?
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 5 — Definisjon: Effekt som kraft ganger fart
%--------------------------------------------------------------------------
\begin{frame}{}

  Effekt kan også uttrykkes ved kraft og fart. Hvis en konstant kraft $F$ virker på et objekt som beveger seg med fart $v$:

  \pause
  \defnat{5.8}{Effekt som kraft ganger fart}{
  \[
    P = F\cdot v
  \]
  }

  \pause
  \merk{Praktisk tolkning}{
  Jo større kraft eller fart, desto mer arbeid utføres per tidsenhet --- og desto større effekt.
  }

\end{frame}

%  SLIDE 6 — Aktivitet: heis

\begin{frame}{Aktivitet}

  En heismotor løfter en heis med total masse $800\,\mathrm{kg}$ med konstant fart $1.5\,\mathrm{m/s}$.
  \begin{enumerate}[a)]
    \item Hvor stor kraft må motoren minst utøve (konstant fart, ingen friksjon)?
    \item Hva er effekten til motoren?
  \end{enumerate}
  }

\end{frame}

  Hvor stor effekt må rytteren yte for å holde farten?
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 8 — Oppsummering: effekt
%--------------------------------------------------------------------------
\begin{frame}{}

  \oppsum{Effekt}{%
  \begin{enumerate}[1.]
    \item[] $P = W/t$ --- arbeid (eller energi overført) per tidsenhet.
    \item[] $P = F\cdot v$ --- for konstant kraft og fart.
    \item[] Enhet: watt ($\mathrm{W}$), $1\,\mathrm{W}=1\,\mathrm{J/s}$.
  \end{enumerate}
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 10 — Til ettertanke: elastisk støt
%--------------------------------------------------------------------------
\begin{frame}{}

  \tanke{}{%
  Er den kinetiske energien alltid bevart i en støt-prosess?
  }
  \pause
  \svar{
  Nei. I et \textit{elastisk støt} er både bevegelsesmengde og kinetisk energi bevart. I et \textit{uelastisk støt} er bare bevegelsesmengden bevart, mens noe av den kinetiske energien omformes til varme, lyd eller deformasjon.
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 11 — Definisjon: Elastisk støt
%--------------------------------------------------------------------------
\begin{frame}{Elastisk støt}

  \defnat{5.9}{Elastisk støt}{
  Et elastisk støt er en kollisjon der \textit{både} bevegelsesmengden og den kinetiske energien er bevart:
  \[
    m_1 v_1 + m_2 v_2 = m_1 u_1 + m_2 u_2
  \]
  \[
    \frac{1}{2}m_1 v_1^2 + \frac{1}{2}m_2 v_2^2
    = \frac{1}{2}m_1 u_1^2 + \frac{1}{2}m_2 u_2^2
  \]
  Her er $v_1$ og $v_2$ hastighetene før støtet, og $u_1$ og $u_2$ hastighetene etter.
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 12 — Merk: Newtons pendel
%--------------------------------------------------------------------------
\begin{frame}{}

  \merk{Newtons pendel}{
  Et klassisk eksempel: kulene i Newtons pendel overfører bevegelsesmengde og kinetisk energi gjennom hele rekken --- den siste kula ``skytes'' opp med (omtrent) samme fart som den første ble sluppet med.
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 13 — Aktivitet: elastisk kollisjon mellom vogner med ulik masse
%--------------------------------------------------------------------------
\begin{frame}{Aktivitet}

  \aktivitet{Elastisk kollisjon mellom to vogner}{%
  En vogn med masse $m_1=2.0\,\mathrm{kg}$ og fart $v_1=3.0\,\mathrm{m/s}$ krasjer inn i en vogn med masse $m_2=1.0\,\mathrm{kg}$ som står i ro ($v_2 = 0$), på et friksjonsfritt underlag. Kollisjonen er elastisk.

  Finn hastighetene $u_1$ og $u_2$ til de to vognene etter kollisjonen.
  }
  \textit{Tips:} Sett opp bevaring av bevegelsesmengde og bevaring av kinetisk energi. Da har du to likninger med to ukjente.

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 14 — Aktivitet: to biljardkuler
%--------------------------------------------------------------------------
\begin{frame}{Aktivitet}

  \aktivitet{To biljardkuler}{%
  En biljardkule ($m=0.17\,\mathrm{kg}$) med fart $v_1=2.0\,\mathrm{m/s}$ treffer en identisk kule i ro, midt i sentrum (rett front-mot-front).

  Hva skjer med de to kulene etter støtet? Løs på samme måte som i forrige aktivitet, men nå med like masser.
  }

\end{frame}

%--------------------------------------------------------------------------
%  SLIDE 15 — Oppsummering
%--------------------------------------------------------------------------
\begin{frame}{}

  \oppsum{Elastisk støt}{%
  \begin{enumerate}[1.]
    \item[] Bevegelsesmengde \textit{og} kinetisk energi er bevart.
    \item[] To likninger, to ukjente --- løses samtidig.
    \item[] Ved kollisjon mellom \textit{like} masser, der én er i ro: farten overføres fullstendig.
  \end{enumerate}
  }

\end{frame}
