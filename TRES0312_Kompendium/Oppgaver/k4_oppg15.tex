\begin{enumerate}[a)]
    \item Tegn kreftene på klossen og dekomponer tyngdekraften langs og vinkelrett på skråplanet.
    \item Finn normalkraften $N$.
    \item Finn friksjonskraften $R$.
    \item Finn akselerasjonen til klossen.
\end{enumerate}
